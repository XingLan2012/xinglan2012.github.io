# 线段树（区间维护类问题的通解）
线段树的核心思想：分治

线段树的树形结构是一种二叉树
去掉最后一层的情况下，它一定是一棵满二叉树
最后一层长得很像完全二叉树，但不是完全二叉树

所以在建树时我们往往将线段树当成满二叉树

不需要建图就可以存储树形结构的方法
1. 满二叉树，节点 $i$ 的左右孩子分别为 $2*i$ 和 $2*i+1$
2. 存 $father[i]$

一个包含 $n$ 个节点的数组，用线段树维护需要使用多大空间：$4n$

将 $[1,n]$ 用二分的方式拆开区间，以满二叉树的形态存储，最多需要用到几个节点
 
一共 $n$ 个叶子，所以 $high = log_2n+1$
满二叉树的节点数量是 $2^0+2^1+2^2+2^3 \dots 2^{high}=2^{high+1}-1=2^{log_2n+2}-1=4 * 2^{log_2n}-1=4*n-1$

```cpp
void pushup(int id){
	sumv[id] = sumv[id * 2] + sumv[id * 2 + 1];
	maxv[id] = max(maxv[id * 2], maxv[id * 2 + 1]);
	minv[id] = min(minv[id * 2], minv[id * 2 + 1]);
}
void pushdown(int id, int l, int r){
	if (lazy[id] != 0){
		int mid = (l + r) / 2;
		//左区间 id * 2 [l, mid] + lazy[id]
		sumv[id * 2] += (mid - l + 1) * lazy[id];
		maxv[id * 2] += lazy[id];
		minv[id * 2] += lazy[id];
		lazy[id * 2] += lazy[id];
		//右区间 id * 2 [mid + 1, r] + lazy[id]	
		sumv[id * 2 + 1] += (r - mid) * lazy[id];
		maxv[id * 2 + 1] += lazy[id];
		minv[id * 2 + 1] += lazy[id];
		lazy[id * 2 + 1] += lazy[id];
		//清空id
		lazy[id] = 0;
	}
}
//build(1,1,n)建树
void build(int id, int l, int r){
	//更新到叶子节点结束
	if (l == r){
		sumv[id] = a[l];
		maxv[id] = a[l];
		minv[id] = a[l];
		return;
	}
	int mid = (l + r) / 2;
	build(id * 2, l, mid);
	build(id * 2 + 1, mid + 1, r);
	pushup(id);
}
//update_point(1, 1, n, x, v) 单点更新 a[x] = v 
void update_point(int id, int l, int r, int x, int v){
	//单点更新直到叶子结点
	if (l == r){
		a[l] = v;
		sumv[id] = a[l];
		maxv[id] = a[l];
		minv[id] = a[l];
		return;
	}
	pushdown(id, l, r);
	int mid = (l + r) / 2;
	if (x <= mid){	
		update_point(id * 2, l, mid, x, v);
	} else {
		update_point(id * 2 + 1, mid + 1, r, x, v);
	}
	pushup(id);
}
// update_range(1, 1, n, L, R, v) 区间 [L,R] + v
void update_range(int id, int l, int r, int L, int R, int v){
	if (L <= l && r <= R){
		lazy[id] += v;
		sumv[id] += (r - l + 1) * v;
		maxv[id] += v;
		minv[id] += v;
		return;	
	}
	pushdown(id, l, r);
	int mid = (l + r) / 2;
	if (L <= mid){
		update_range(id * 2, l, mid, L, R, v);
	} 
	if (R > mid){
		update_range(id * 2 + 1, mid + 1, r, L, R, v);	
	}
	pushup(id);
}
// query_range(1, 1, n, L, R)区间求和 [L,R]
void query_range_sumv(int id, int l, int r, int L, int R){
	if (L <= l && r <= R){
		return sumv[id];
	}
	int mid = (l + r) / 2;
	int ans = 0;
	pushdown(id, l, r);
	if (L <= mid){
		ans += query_range_sumv(id * 2, l, mid, L, R);
	} 
	if (R > mid){
		ans += query_range_sumv(id * 2 + 1, mid + 1, r, L, R);
	}
	return ans;
}
```

### 总结
线段树相关的题目考察点一般只有两个
1. 线段树的区间操作较为复杂
2. 维护和查询的内容是不一样的

多想，多画，慢慢写

|名称|适用性|初始化|区间查询|更新|
|:---:|:---:|:---:|:---:|:---:|
|ST表|静态|$O(nlogn)$|$O(1)$|重新初始化|
|树状数组|动态前缀和|$O(nlogn)$|$O(logn)$|单点更新$O(logn)$|
|线段树|通解|$O(n)$|$O(logn)$|区间更新$O(logn)$|

注意事项：
1. 树状数组：代码短，常数小，但是适用性不强，区间修改难度较高
2. 线段树：功能性强，代码稍长，**常数大**

### 棋子等级
由于给定的棋子顺序是按 $Y$ 排序后按 $X$ 排序的
所以对于第 $i$ 个棋子需要计算的其实就是 $1 \sim i$ 中有多少 $x <= x_i$

使用 $cnt[x[i]]++$ 来记录坐标为 $x$ 的棋子有多少个
对于 $x_i$ 就是查询 $cnt[1] \sim cnt[x[i]]$ 的和

用线段树/树状数组维护 $cnt$ 数组即可


### 旅馆
房间只有 $0/1$ 用于标记是否有人
每次查询最左侧连续 $x$ 个 $0$ 的位置，然后将它标记为 $1$

对于查询，我们只要优先考虑左区间即可

特殊情况：左右两个子区间合并时，中间会出现一段连续的 $0$

答案存在三种情况：左区间，中间，右区间

显然我们至少需要知道的信息是：某个区间内最长连续的空房间数量
由于两个子区间合并以后，中间是可以拼接的，那么我们需要额外记录
区间内左端点连续空房间数量，右端点连续空房间数量

```cpp
struct Node{
	int val, left, right, lazy;
} tree[400010];

void pushup(int id, int l, int r){
	int ls = id * 2;
	int rs = id * 2 + 1;
	int mid = (l + r) / 2;
	
	tree[id].left = tree[ls].left;
	if (tree[ls].left == mid - l + 1){
		tree[id].left += tree[rs].left;
	}
	tree[id].right = tree[rs].right;
	if (tree[rs].right == r - mid){
		tree[id].right += tree[ls].right;	
	}
	
	tree[id].val = max(tree[ls].val, tree[rs].val);
	tree[id].val = max(tree[id].val, tree[ls].right + tree[rs].left);
}
void pushdown(int id, int l, int r){
	if (tree[id].lazy == 0){
		tree[id * 2].val = mid - l + 1;
		tree[id * 2].left = mid - l + 1;
		tree[id * 2].right = mid - l + 1;
		tree[id * 2].lazy = 0;
		
		tree[id * 2 + 1].val = r - mid;
		tree[id * 2 + 1].left = r - mid;
		tree[id * 2 + 1].right = r - mid;
		tree[id * 2 + 1].lazy = 0;
		
		tree[id].lazy = -1;
	} else if (tree[id].lazy == 1){
		tree[id * 2].val = 0;
		tree[id * 2].left = 0;
		tree[id * 2].right = 0;
		tree[id * 2].lazy = 1;
		
		tree[id * 2 + 1].val = 0;
		tree[id * 2 + 1].left = 0;
		tree[id * 2 + 1].right = 0;
		tree[id * 2 + 1].lazy = 1;
		
		tree[id].lazy = -1;
	}
}
void build(int id, int l, int r){
	tree[id].lazy = -1;
	if (l == r){
		tree[id].left = tree[id].right = tree[id].val = 1;
		return;	
	}
	int mid = (l + r) / 2;
	build(id * 2, l, mid);
	build(id * 2 + 1, mid + 1, r);
	pushup(id, l, r);
}
void update(int id, int l, int r, int L, int R, int x){
	if (L <= l && r <= R){
		if (x == 0){
			tree[id].val = r - l + 1;
			tree[id].left = r - l + 1;
			tree[id].right = r - l + 1;
			tree[id].lazy = 0;
		} else {
			tree[id].val = 0;
			tree[id].left = 0;
			tree[id].right = 0;
			tree[id].lazy = 1;
		}
		return;	
	}
	pushdown(id, l, inr r);
	int mid = (l + r) / 2;
	if (L <= mid){
		update(id * 2, l, mid, L, R, x);	
	}
	if (R > mid){
		update(id * 2 + 1, mid + 1, r, L, R, x);	
	}
	pushup(id, l, r);
}
int query(int id, int l, int r, int x){
	if (l == r){
		return l;	
	}
	pushdown(id, l, r);
	int mid = (l + r) / 2;
	if (tree[id * 2].val >= x){
		return query(id * 2, l, mid, x);
	}
	if (tree[id * 2].right + tree[id * 2 + 1].left >= x){
		return mid - tree[id * 2].right + 1;
	}
	return query(id * 2 + 1, mid + 1, r, x);
}
```

### 线段树动态开点
线段树由于需要当成满二叉树来构造，所以空间消耗过大
但是很多时候，如果区间很大，但是利用的节点很少
1. 离散化（离线）
2. 强制在线，动态开点

棋子等级plus：$x,y$ 的范围有 $1e8$，强制在线

线段树动态开点：假设这棵树 **虚空存在**，然后跟 $lazy$ 思想一样，当碰到一个节点它还不存在的时候，新建一个节点给它，那么我们需要的空间是由访问过的节点数量决定的

一次操作最多产生 $4logn$ 个节点
最多产生 $m * 4logn$ 个节点

由于动态产生节点，那么就不能无脑用 $i$ 和 $i*2,i*2+1$ 来表示父子关系
那么就采用链表的构造方式：用结构体 $lson,rson$ 记录左儿子节点编号，右儿子节点编号
每次递归碰到不存在的节点时候，分配 $cnt++$ 给它

```cpp
struct Node{
	int sumv, minv, maxv, lson, rson;
}
int query_min(int &id, int l, int r, int L, int R){
	//表示节点不存在
	if (id == -1){
		//新建一个节点赋值给 id
		cnt++;
		id = cnt;
		//节点初始化
		tree[id].sumv = (l + r) * (r - l + 1) / 2;
		tree[id].minv = l;
		tree[id].maxv = r;
		tree[id].lson = tree[id].rson = -1;//新建的节点左右儿子均不存在
		//假设 a 数组初始值 a[i] = i;
	}
	if (L <= l && r <= R){
		return minv[id];	
	}
	pushdown(id, l, r);
	int mid = (l + r) / 2;
	int ans = INF;
	if (L <= mid){
		ans = min(ans, query_min(tree[id].lson, l, mid, L, R);	
	}	
	if (R > mid){
		ans = min(ans, query_min(tree[id].lson, mid + 1, r, L, R);	
	}
	return ans;
}
```
